function param(value: string | string[] | undefined) {
  return Array.isArray(value) ? value[0] : value ?? '';
}

function intParam(value: string | string[] | undefined, fallback: number) {
  const raw = param(value);
  if (!raw.trim()) return fallback;
  const parsed = Number(raw);
  return Number.isFinite(parsed) ? Math.round(parsed) : fallback;
}

export default function FocusScreen() {
  const params = useLocalSearchParams<{
    topicId?: string | string[];
    taskId?: string | string[];
    minutes?: string | string[];
    breakMinutes?: string | string[];
  }>();

  const { width: windowWidth } = useWindowDimensions();
  // Reserve content padding and the outer-ring overshoot on small screens.
  const timerDialSize = focusDialSize(windowWidth, spacing.lg);
  const { workspace, ownerKey } = useWorkspace();
  const {
    ready,
    session,
    saveError,
    start,
    pause,
    resume,
    extendLaunch,
    park,
    removeParked,
    completeFocus,
    completeBreak,
    skipBreak,
    discard,
  } = useFocus();
  const { session: activeReview, ready: reviewReady } = useReview();
  const { ready: onboardingReady, shouldShow: onboardingPending } = useOnboarding();
  const { transitionPending } = useOnboardingTransition();

  const topicId = param(params.topicId);
  const taskId = param(params.taskId);
  const task = workspace && taskId ? workspace.tasks.find((item) => item.id === taskId) ?? null : null;
  const topic = workspace ? findTopic(workspace, task?.topicId || topicId) : null;
  const suggestedMinutes = Math.max(
    5,
    Math.min(240, intParam(params.minutes, workspace?.settings.focusMinutes ?? 45)),
  );
  const breakMinutes = Math.max(
    1,
    Math.min(30, intParam(params.breakMinutes, workspace?.settings.breakMinutes ?? 10)),
  );

  const [selectedMinutes, setSelectedMinutes] = useState<number | null>(null);
  const plannedMinutes = selectedMinutes ?? suggestedMinutes;
  const [savingFocus, setSavingFocus] = useState(false);
  const [focusError, setFocusError] = useState('');
  const [tinyStep, setTinyStep] = useState('');
  const [distraction, setDistraction] = useState('');
  const [showParking, setShowParking] = useState(false);
  const [now, setNow] = useState(Date.now());
  const [appIsActive, setAppIsActive] = useState(
    AppState.currentState !== 'background' && AppState.currentState !== 'inactive',
  );
  const [finishing, setFinishing] = useState(false);
  const [effort, setEffort] = useState<1 | 2 | 3 | 4 | 5>(3);
  const [breakTransition, setBreakTransition] = useState<'finish' | 'retrieve' | 'skip' | null>(null);
  const breakTransitionLock = useRef(false);

  // Android suspends JS intervals while the app is backgrounded. The timer
  // itself stays anchored to persisted timestamps, never to interval ticks.
  // Snap the display to wall-clock time as soon as MYELIN is foregrounded.
  useEffect(() => {
    const listener = AppState.addEventListener('change', (nextState) => {
      const active = nextState === 'active';
      setAppIsActive(active);
      if (active) setNow(Date.now());
    });
    return () => listener.remove();
  }, []);

  // The screen may stay mounted while the user visits another in-app route.
  // Refresh on navigation focus too, without changing the running session.
  useFocusEffect(useCallback(() => {
    setNow(Date.now());
  }, []));

  useEffect(() => {
    if (session?.runningSince == null || !appIsActive) return;
    setNow(Date.now());
    const timer = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(timer);
  }, [appIsActive, session?.runningSince, session?.phase]);

  useEffect(() => {
    setNow(Date.now());
  }, [session?.id, session?.phase, session?.plannedFocusMinutes]);

  // Release a completed break's interaction lock when the persisted session
  // changes phase or a different focus session begins.
  useEffect(() => {
    if (session?.phase === 'break') return;
    breakTransitionLock.current = false;
    setBreakTransition(null);
  }, [session?.id, session?.phase]);

  const remaining = useMemo(
    () => (session ? timerRemainingMs(session, now) : 0),
    [session, now],
  );
  const timerDone = session ? timerIsComplete(session, now) : false;
  const phaseDurationMs = session
    ? (session.phase === 'break' ? session.breakMinutes : session.plannedFocusMinutes) * 60_000
    : 1;
  const phaseProgress = session
    ? Math.max(0, Math.min(1, 1 - remaining / Math.max(1, phaseDurationMs)))
    : 0;
  const retrievalSuggestion = useMemo(
    () =>
      workspace && session?.phase === 'break'
        ? buildPostFocusRetrievalSuggestion(
            workspace,
            session.topicId,
            session.focusActualMinutes,
            now,
          )
        : null,
    [now, session?.focusActualMinutes, session?.phase, session?.topicId, workspace],
  );
  const canExtend = !!session && session.phase === 'focus' && !finishing &&
    session.launchMode === 'five-minute' &&
    session.fullTargetMinutes > session.plannedFocusMinutes && timerDone;
  const guideState: FocusGuideState = session
    ? session.phase === 'break' ? timerDone ? 'break-done' : 'break'
      : canExtend ? 'checkpoint' : finishing || timerDone ? 'finish' : 'active'
    : activeReview ? 'review' : (!topic && !task) || task?.status === 'completed' ? 'empty' : 'setup';
  const guideReady = ready && reviewReady && !!workspace && !savingFocus && !breakTransition;
  const guide = useFocusGuide(guideReady, ownerKey, guideState);
  const guideOffer = useGuideOffer('focus', ownerKey,
    guideReady && guideState === 'setup' && onboardingReady && !onboardingPending && !transitionPending,
    guide.open && guideState === 'setup');
  const guideOverlay = guide.open ? <ScreenGuide key={guideState} title="Focus" steps={guide.steps}
    anchors={guide.anchors} revealAnchor={guide.revealAnchor} onClose={guide.close} /> : null;

  async function begin(launchMode: 'full' | 'five-minute', firstStep = tinyStep) {
    if (!topic && !task) return;
    setFocusError('');
    try {
      const next = await start({
        ...(topic ? { topicId: topic.id } : {}),
        ...(task ? { taskId: task.id } : {}),
        plannedMinutes,
        breakMinutes,
        launchMode,
        tinyStep: firstStep,
      });
      if (next) {
        setNow(Date.now());
        setFinishing(false);
      }
    } catch {
      setFocusError('Could not save the Focus session. Check device storage and try starting again.');
    }
  }

  async function addDistraction() {
    const value = distraction.trim();
    if (!value) return;
    setFocusError('');
    try {
      await park(value);
      setDistraction('');
    } catch {
      setFocusError('The thought could not be saved. Your text is still here; try again.');
    }
  }

  function requestFinishEarly() {
    setFinishing(true);
    void pause();
  }

  function requestDiscard() {
    Alert.alert(
      'Discard this focus session?',
      'The timer and parked distractions will be removed. No study time will be added to history.',
      [
        { text: 'Keep session', style: 'cancel' },
        {
          text: 'Discard',
          style: 'destructive',
          onPress: async () => {
            try {
              setFocusError('');
              await discard();
              router.replace('/(app)/today');
            } catch {
              setFocusError('Could not clear this session. Stay here and retry.');
            }
          },
        },
      ],
    );
  }

  async function finishFocus() {
    if (savingFocus) return;
    setSavingFocus(true); setFocusError('');
    try {
      await completeFocus(effort, Date.now());
      setFinishing(false); setNow(Date.now());
    } catch { setFocusError('Your session could not be saved. Tap Complete focus to retry.'); }
    finally { setSavingFocus(false); }
  }

  async function runBreakTransition(action: 'finish' | 'retrieve' | 'skip') {
    // Use an immediate ref guard: state-only disabling does not stop two
    // rapid presses before React has rendered the disabled button.
    if (breakTransitionLock.current || session?.phase !== 'break') return;
    breakTransitionLock.current = true;
    setBreakTransition(action);
    setFocusError('');

    try {
      // Snapshot the optional retrieval destination before clearing the break.
      const suggestion = action === 'retrieve' ? retrievalSuggestion : null;
      if (action === 'skip') {
        await skipBreak(Date.now());
      } else {
        await completeBreak(Date.now());
      }

      if (suggestion) {
        router.replace({
          pathname: '/(app)/review',
          params: {
            topicId: suggestion.topicId,
            autoStart: '1',
            limit: String(suggestion.count),
            source: 'focus',
          },
        });
      } else {
        router.replace('/(app)/today');
      }
      // Keep the action disabled through navigation, not just the save.
    } catch {
      breakTransitionLock.current = false;
      setBreakTransition(null);
      setFocusError('Break progress could not be saved. Stay here and retry.');
    }
  }

  if (!ready || !workspace) {
    return (
      <SafeAreaView style={styles.safe}>
        <View style={styles.center}>
          <Text style={styles.loading}>Restoring focus session…</Text>
        </View>
      </SafeAreaView>
    );
  }

  if (!session && activeReview) {
    return (
      <SafeAreaView style={styles.safe}>
        <AmbientBackdrop />
        <View style={styles.setupPage}>
          <TopBar onHelp={guide.start} label="Focus" backLabel="Review" onBack={() => router.replace('/(app)/review')} />
          <View style={styles.missing}>
            <Text style={styles.missingTitle}>A review queue is already active.</Text>
            <Text style={styles.missingText}>
              Resume or end that recall queue before starting a separate focus timer.
            </Text>
            <View ref={guide.anchors['focus.primary']} collapsable={false} testID="guide-focus-primary"><Pressable
              accessibilityRole="button"
              onPress={() => router.replace('/(app)/review')}
              style={({ pressed }) => [styles.primary, pressed && styles.pressed]}>
              <Text style={styles.primaryText}>Resume review</Text>
              <Text style={styles.buttonArrow}>→</Text>
            </Pressable></View>
          </View>
        </View>
        {guideOverlay}
      </SafeAreaView>
    );
  }

  if (!session) {
    if ((!topic && !task) || (task && task.status !== 'open')) {
      return (
        <SafeAreaView style={styles.safe}>
          <AmbientBackdrop />
          <View style={styles.setupPage}>
            <TopBar onHelp={guide.start} label="Focus" onBack={() => router.replace('/(app)/today')} />
            <View style={styles.missing}>
              <Text style={styles.missingTitle}>Choose an open task or topic.</Text>
              <Text style={styles.missingText}>
                Start Focus from any task card or from a topic in Library.
              </Text>
              <View ref={guide.anchors['focus.primary']} collapsable={false} testID="guide-focus-primary"><Pressable
                accessibilityRole="button"
                onPress={() => router.replace('/(app)/today')}
                style={({ pressed }) => [styles.primary, pressed && styles.pressed]}>
                <Text style={styles.primaryText}>Back to Today</Text>
              </Pressable></View>
            </View>
          </View>
          {guideOverlay}
        </SafeAreaView>
      );
    }

    return (
      <SafeAreaView style={styles.safe}>
        <AmbientBackdrop />
        <View style={styles.setupPage}>
          <TopBar onHelp={guide.start} label="Focus" backLabel="Back" onBack={() => router.back()} />

          <ScrollView
            ref={guide.scroll} onScroll={event => { guide.offset.current = event.nativeEvent.contentOffset.y; }} scrollEventThrottle={16}
            keyboardShouldPersistTaps="handled"
            contentContainerStyle={styles.setupContent}
            showsVerticalScrollIndicator={false}>
            <Text style={styles.eyebrow}>{(task?.subject || topic?.subject || 'STUDY TASK').toUpperCase()}</Text>
            <Text style={styles.setupTitle}>{task?.title || topic?.name}</Text>
            {task ? (
              <Text style={styles.taskContext}>{topic ? `Linked topic · ${topic.name}` : 'Time is saved to this task and today’s used time'}</Text>
            ) : null}
            <Text style={styles.setupSubtitle}>
              Choose the first action, then let the timer disappear into the background.
            </Text>

            <FocusSupport key={task?.id ?? topic?.id} task={task} topicPostponements={topic?.postponements} onStep={setTinyStep} onLaunch={step => begin('five-minute', step)} />
            <View style={styles.planStrip}>
              <Metric label="FULL BLOCK" value={`${plannedMinutes} min`} />
              <Metric label="BREAK" value={`${breakMinutes} min`} />
            </View>

            <View ref={guide.anchors['focus.primary']} collapsable={false} testID="guide-focus-primary" style={{ flexDirection: 'row', flexWrap: 'wrap', gap: 8, marginTop: 12 }}>
              {[10, 25, 45].map(minutes => <Pressable key={minutes} accessibilityRole="button"
                accessibilityLabel={`Focus for ${minutes} minutes`} accessibilityState={{ selected: plannedMinutes === minutes }}
                onPress={() => setSelectedMinutes(minutes)} style={[styles.secondary, { minWidth: 64 }, plannedMinutes === minutes && styles.effortActive]}>
                <Text style={styles.secondaryText}>{minutes} min</Text>
              </Pressable>)}
            </View>
            <Text style={styles.fieldLabel}>First tiny step · optional</Text>
            <View ref={guide.anchors['focus.secondary']} collapsable={false} testID="guide-focus-secondary"><TextInput
              accessibilityLabel="First tiny step"
              value={tinyStep}
              onChangeText={setTinyStep}
              placeholder="e.g. Solve question 1 without notes"
              placeholderTextColor={colors.subtle}
              selectionColor={colors.accent}
              maxLength={500}
              multiline
              textAlignVertical="top"
              style={styles.tinyInput}
            /></View>

            <Pressable
              accessibilityRole="button"
              onPress={() => begin('full')}
              style={({ pressed }) => [styles.primary, pressed && styles.pressed]}>
              <View style={{ flex: 1, minWidth: 0 }}>
                <Text style={styles.primaryText}>Start {plannedMinutes}-minute focus</Text>
                <Text style={styles.primarySubtext}>One task. A little more progress.</Text>
              </View>
              <Text style={styles.buttonArrow}>→</Text>
            </Pressable>

            <View ref={guide.anchors['focus.support']} collapsable={false} testID="guide-focus-support"><Pressable
              accessibilityRole="button"
              onPress={() => begin('five-minute')}
              style={({ pressed }) => [styles.launchButton, pressed && styles.pressed]}>
              <View style={styles.launchMark}>
                <Text style={styles.launchMarkText}>5</Text>
              </View>
              <View style={styles.launchCopy}>
                <Text style={styles.launchTitle}>Five-minute launch</Text>
                <Text style={styles.launchText}>
                  Only promise five minutes. If momentum arrives, continue into the full block.
                </Text>
              </View>
            </Pressable></View>

            <GuideOffer tutorialId="focus" screenName="Focus"
              description="See block length, your first tiny step and the five-minute launch. Replay anytime from ?."
              visible={guideOffer.visible} saveFailed={guideOffer.saveFailed}
              onShow={() => { guideOffer.choose('shown'); guide.start(); }}
              onLater={() => guideOffer.choose('later')} onRetry={guideOffer.retry} />

            <DistractionShield />
            {focusError || saveError ? (
              <Text accessibilityRole="alert" style={styles.storageWarning}>{focusError || saveError}</Text>
            ) : null}
            <Text style={styles.recoveryNote}>
              The timer survives leaving or closing MYELIN.
            </Text>
          </ScrollView>
        </View>
        {guideOverlay}
      </SafeAreaView>
    );
  }

  if (session.phase === 'break') {
    return (
      <SafeAreaView style={styles.safe}>
        <AmbientBackdrop />
        <View style={styles.sessionPage}>
          <TopBar
            onHelp={guide.start}
            label="Break"
            onBack={() => router.replace('/(app)/today')}
            helper={timerDone ? 'Timer complete' : session.runningSince ? 'Timer stays active' : 'Timer paused'}
          />

          <ScrollView
            ref={guide.scroll} onScroll={event => { guide.offset.current = event.nativeEvent.contentOffset.y; }} scrollEventThrottle={16}
            contentContainerStyle={styles.timerArea}
            showsVerticalScrollIndicator={false}>
            <Text style={styles.eyebrow}>RECOVERY</Text>
            <Text style={styles.sessionTitle}>Leave the work alone.</Text>
            {session.focusLogged ? (
              <Reveal key={`focus-saved:${session.id}`} distance={5}>
                <Text style={styles.focusSavedCue}>
                  ✓ Focus logged · {session.focusActualMinutes} min
                </Text>
              </Reveal>
            ) : null}

            <TimerDial
              value={formatTimer(remaining)}
              label={timerDone ? 'BREAK COMPLETE' : session.runningSince ? 'BREAK' : 'PAUSED'}
              complete={timerDone}
              progress={phaseProgress}
              size={timerDialSize}
            />
            <View style={styles.phaseProgress}>
              <AnimatedProgress
                value={phaseProgress}
                accessibilityLabel={`Break progress: ${Math.round(phaseProgress * 100)}%`}
              />
            </View>

            {focusError || saveError ? (
              <Text accessibilityRole="alert" style={styles.storageWarning}>{focusError || saveError}</Text>
            ) : null}
            <Text style={styles.breakPrompt}>
              {session.focusActualMinutes >= 25 ? 'Step away from the screen. Walk a little, stretch gently, or get water.' : 'Let your eyes rest on something far away. Take a few unhurried breaths.'}
              {session.distractions.length ? ` You parked ${session.distractions.length} thought${session.distractions.length === 1 ? '' : 's'}. Check them before leaving this break.` : ' You do not need to earn the break.'}
            </Text>
            {session.distractions.length ? <ScrollView nestedScrollEnabled style={{ maxHeight: 120, flexShrink: 0, alignSelf: 'stretch' }}>{session.distractions.map(item => <Text key={item.id} style={styles.breakPrompt}>• {item.text}</Text>)}</ScrollView> : null}

            {retrievalSuggestion ? (
              <View style={styles.retrieveCue}>
                <Text style={styles.retrieveCueKicker}>AFTER THE BREAK · RETRIEVE</Text>
                <Text style={styles.retrieveCueTitle}>
                  {retrievalSuggestion.count} quick recall{' '}
                  {retrievalSuggestion.count === 1 ? 'prompt' : 'prompts'} ·{' '}
                  {retrievalSuggestion.topicName}
                </Text>
                <Text style={styles.retrieveCueText}>
                  These are already ready in Memory Engine. MYELIN will not pull
                  future cards forward just to fill this step.
                </Text>
              </View>
            ) : null}

            <View ref={guide.anchors['focus.primary']} collapsable={false} testID="guide-focus-primary" style={{ alignSelf: 'stretch' }}>
            <Reveal key={timerDone ? 'break-done' : 'break-running'} distance={6} style={styles.breakActions}>
              {timerDone ? (
                retrievalSuggestion ? (
                  <View style={styles.retrieveActions}>
                    <Pressable
                      accessibilityRole="button"
                      accessibilityLabel={`Finish break and retrieve ${retrievalSuggestion.count} recall prompts`}
                      accessibilityState={{ disabled: Boolean(breakTransition), busy: Boolean(breakTransition) }}
                      disabled={Boolean(breakTransition)}
                      onPress={() => void runBreakTransition('retrieve')}
                      style={({ pressed }) => [
                        styles.primary,
                        styles.wideButton,
                        breakTransition && styles.disabled,
                        pressed && !breakTransition && styles.pressed,
                      ]}>
                      <View style={styles.retrieveActionCopy}>
                        <Text style={styles.primaryText}>
                          {breakTransition === 'retrieve' ? 'Saving break…' : (
                            <>
                              Retrieve {retrievalSuggestion.count}{' '}
                              {retrievalSuggestion.count === 1 ? 'prompt' : 'prompts'}
                            </>
                          )}
                        </Text>
                        <Text style={styles.primarySubtext}>
                          Short recall, then back to Today
                        </Text>
                      </View>
                      <Text style={styles.buttonArrow}>→</Text>
                    </Pressable>
                    <Pressable
                      accessibilityRole="button"
                      accessibilityState={{ disabled: Boolean(breakTransition), busy: Boolean(breakTransition) }}
                      disabled={Boolean(breakTransition)}
                      onPress={() => void runBreakTransition('finish')}
                      style={({ pressed }) => [
                        styles.textButton,
                        breakTransition && styles.disabled,
                        pressed && !breakTransition && styles.pressed,
                      ]}>
                      <Text style={styles.textButtonText}>
                        {breakTransition === 'finish' ? 'Saving break…' : 'Finish break without recall'}
                      </Text>
                    </Pressable>
                  </View>
                ) : (
                  <Pressable
                    accessibilityRole="button"
                    accessibilityState={{ disabled: Boolean(breakTransition), busy: Boolean(breakTransition) }}
                    disabled={Boolean(breakTransition)}
                    onPress={() => void runBreakTransition('finish')}
                    style={({ pressed }) => [
                      styles.primary,
                      styles.wideButton,
                      breakTransition && styles.disabled,
                      pressed && !breakTransition && styles.pressed,
                    ]}>
                    <Text style={styles.primaryText}>
                      {breakTransition === 'finish' ? 'Saving break…' : 'Finish break'}
                    </Text>
                    <Text style={styles.buttonArrow}>→</Text>
                  </Pressable>
                )
              ) : (
                <View style={styles.timerActions}>
                  <Pressable
                    accessibilityRole="button"
                    onPress={session.runningSince ? pause : resume}
                    style={({ pressed }) => [styles.secondary, pressed && styles.pressed]}>
                    <Text style={styles.secondaryText}>{session.runningSince ? 'Pause' : 'Resume'}</Text>
                  </Pressable>
                  <Pressable
                    accessibilityRole="button"
                    accessibilityState={{ disabled: Boolean(breakTransition), busy: Boolean(breakTransition) }}
                    disabled={Boolean(breakTransition)}
                    onPress={() => void runBreakTransition('skip')}
                    style={({ pressed }) => [
                      styles.secondary,
                      breakTransition && styles.disabled,
                      pressed && !breakTransition && styles.pressed,
                    ]}>
                    <Text style={styles.secondaryText}>
                      {breakTransition === 'skip' ? 'Saving…' : 'End break'}
                    </Text>
                  </Pressable>
                </View>
              )}
            </Reveal>
            </View>
          </ScrollView>
        </View>
        {guideOverlay}
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.safe}>
      <AmbientBackdrop />
      <View style={styles.sessionPage}>
        <TopBar
          onHelp={guide.start}
          label="Focus"
          onBack={() => router.replace('/(app)/today')}
          helper={timerDone ? 'Timer complete' : session.runningSince ? 'Timer stays active' : 'Timer paused'}
        />

        <ScrollView

const sameSteps = (left: string[], right: string[]) => left.length === right.length && left.every((step, index) => step === right[index]);

function FocusSupportEditor({ task, topicPostponements = 0, onStep, onLaunch }: SupportProps) {

    if (!experiment || !avoided || !task || task.launchTrial || !ownerKey || !ready()) return;
    updateWorkspace(d => {
      const current = d.tasks.find(item => item.id === task.id)!;
      current.launchTrial = { variant: launchVariant(current, true), offeredAt: Date.now(), startedAt: 0 };
    }, { ownerKey, canApply: current => current.settings.launchExperiment === true && current.tasks.some(item => item.id === task.id && item.status === 'open' && (item.postponements ?? 0) >= 2 && !item.launchTrial) });

    const current = ready()?.tasks.find(item => item.id === task.id);
    if (!current || current.status !== 'open') return;
    baseline.current = [...(current.steps ?? [])]; setDraft(suggestedSteps(current).join('\n')); setMessage('Latest steps loaded. Your other task data was kept.');
    submitted.current = false; generation.current++; setVersion(generation.current);
  }
  function safeTask() {
    if (!validDraft()) return false;
    if (!task) return true;
    const current = ready()?.tasks.find(item => item.id === task.id);
    if (!current || current.status !== 'open') { setMessage('This task is no longer open. Your work was kept.'); return false; }
    if (!sameSteps(current.steps ?? [], baseline.current)) { setMessage('The saved steps changed. Your draft was kept. Use latest steps before continuing.'); return false; }
    return true;
  }
  async function launch() {
    if (!safeTask() || launching.current) return;
    launching.current = true;
    try { await onLaunch(draft.split('\n').map(step => step.trim()).find(Boolean)?.slice(0,500) ?? 'Open the material'); }
    finally { launching.current = false; }
  }
  function saveSteps() {
    if (submitted.current || !safeTask()) return;
    const steps = draft.split('\n').map(step => step.trim().slice(0,500)).filter(Boolean).slice(0,12);
    if (!steps.length) { setMessage('Write at least one small step.'); return; }
    if (task && ownerKey && !sameSteps(steps, baseline.current)) {
      const next = updateWorkspace(d => { const current = d.tasks.find(item => item.id === task.id)!; current.steps = steps; current.updatedAt = Date.now(); }, {
        ownerKey, canApply: current => current.tasks.some(item => item.id === task.id && item.status === 'open' && sameSteps(item.steps ?? [], baseline.current)),
      });
      if (!next) { setMessage('The saved steps changed. Your draft was kept. Use latest steps before continuing.'); return; }
      baseline.current = [...steps];
    }
    submitted.current = true; onStep(steps[0]); setMessage(task ? 'Steps saved. Your first step is ready.' : 'First step selected for this focus session.');
  }
  if (!workspace) return null;
  return <View style={styles.wrap}>
    {sleepingAt(workspace.settings, Date.now()) ? <Text style={styles.copy}>This is your protected sleep window. Today will suggest work after you wake. You can still choose a short session.</Text> : null}
    {avoided ? <View style={styles.callout}><Text style={styles.title}>{variant === 'five-minute' || !experiment ? 'Make the start smaller' : 'Choose one concrete first step'}</Text><Text style={styles.copy}>You have postponed this {task?.postponements ?? topicPostponements} times. Pick one step; the whole task can wait.</Text>{variant === 'five-minute' || !experiment ? <Pressable accessibilityRole="button" onPress={launch} style={styles.button}><Text style={styles.action}>Try just five minutes →</Text></Pressable> : null}</View> : null}
    <Pressable accessibilityRole="button" accessibilityState={{ expanded }} onPress={() => { if (validDraft()) setExpanded(value => !value); }} style={styles.button}><Text style={styles.action}>{expanded ? 'Hide first steps' : 'Break this into small steps'}</Text></Pressable>
    <AnimatedDisclosure visible={expanded}>
      <Text style={styles.copy}>Suggested starting points. Edit one step per line to fit your work.</Text>
      <TextInput accessibilityLabel="Small task steps" multiline value={draft} onChangeText={value => { if (validDraft()) { submitted.current = false; setDraft(value); } }} maxLength={6000} style={styles.input} />
      <Pressable accessibilityRole="button" style={styles.button} onPress={saveSteps}><Text style={styles.action}>{task ? 'Save steps & use first step' : 'Use first step'}</Text></Pressable>
      {conflict ? <Pressable accessibilityRole="button" onPress={useLatest} style={styles.button}><Text style={styles.action}>Use latest steps</Text></Pressable> : null}

    </AnimatedDisclosure>
  </View>;
}

import {
  FOCUS_SESSION_FORMAT,
  FOCUS_SESSION_VERSION,
  FocusLaunchMode,
  FocusSessionState,
  ParkedDistraction,
} from './types';

const MINUTE = 60_000;

function clampMinutes(value: number, min: number, max: number) {
  return Math.max(min, Math.min(max, Math.round(value)));
}

function id(prefix: string, now: number) {
  return `${prefix}_${now.toString(36)}_${Math.random().toString(36).slice(2, 9)}`;
}

export function createFocusSession(input: {
  topicId?: string;
  taskId?: string;
  topicName: string;
  subject: string;
  plannedMinutes: number;
  breakMinutes: number;
  launchMode: FocusLaunchMode;
  tinyStep: string;
  now?: number;
  id?: string;
}): FocusSessionState {
  const now = input.now ?? Date.now();
  const fullTargetMinutes = clampMinutes(input.plannedMinutes, 5, 240);
  const plannedFocusMinutes = input.launchMode === 'five-minute' ? 5 : fullTargetMinutes;

  return {
    format: FOCUS_SESSION_FORMAT,
    version: FOCUS_SESSION_VERSION,
    id: input.id ?? id('focus', now),
    ...(input.topicId ? { topicId: input.topicId } : {}),
    ...(input.taskId ? { taskId: input.taskId } : {}),
    topicName: input.topicName,
    subject: input.subject,
    phase: 'focus',
    launchMode: input.launchMode,
    tinyStep: input.tinyStep.trim(),
    fullTargetMinutes,
    plannedFocusMinutes,
    breakMinutes: clampMinutes(input.breakMinutes, 1, 30),
    startedAt: now,
    phaseStartedAt: now,
    elapsedMs: 0,
    runningSince: now,
    focusLogged: false,
    focusActualMinutes: 0,
    distractions: [],
  };
}

export function phaseDurationMs(session: FocusSessionState) {
  const minutes = session.phase === 'focus' ? session.plannedFocusMinutes : session.breakMinutes;
  return minutes * MINUTE;
}

export function timerElapsedMs(session: FocusSessionState, now = Date.now()) {
  const live = session.runningSince === null ? 0 : Math.max(0, now - session.runningSince);
  return Math.max(0, session.elapsedMs + live);
}

export function timerRemainingMs(session: FocusSessionState, now = Date.now()) {
  return Math.max(0, phaseDurationMs(session) - timerElapsedMs(session, now));
}

export function timerIsComplete(session: FocusSessionState, now = Date.now()) {
  return timerRemainingMs(session, now) <= 0;
}

export function pauseFocusTimer(session: FocusSessionState, now = Date.now()): FocusSessionState {
  if (session.runningSince === null) return session;
  return {
    ...session,
    elapsedMs: Math.min(phaseDurationMs(session), timerElapsedMs(session, now)),
    runningSince: null,
  };
}

export function resumeFocusTimer(session: FocusSessionState, now = Date.now()): FocusSessionState {
  if (session.runningSince !== null || timerIsComplete(session, now)) return session;
  return { ...session, runningSince: now };
}

export function extendFiveMinuteLaunch(
  session: FocusSessionState,
  now = Date.now(),
): FocusSessionState {
  if (
    session.phase !== 'focus' ||
    session.launchMode !== 'five-minute' ||
    session.fullTargetMinutes <= session.plannedFocusMinutes
  ) {
    return session;
  }

  const elapsed = Math.min(phaseDurationMs(session), timerElapsedMs(session, now));

  return {
    ...session,
    launchMode: 'full',
    plannedFocusMinutes: session.fullTargetMinutes,
    elapsedMs: elapsed,
    runningSince: now,
  };
}

export function addParkedDistraction(
  session: FocusSessionState,
  text: string,
  now = Date.now(),
): FocusSessionState {
  const clean = text.trim().slice(0, 240);
  if (!clean || session.distractions.length >= 30) return session;

  const distraction: ParkedDistraction = {
    id: id('park', now),
    text: clean,
    createdAt: now,
  };

  return {
    ...session,
    distractions: [...session.distractions, distraction],
  };
}

export function removeParkedDistraction(
  session: FocusSessionState,
  distractionId: string,
): FocusSessionState {
  return {
    ...session,
    distractions: session.distractions.filter((item) => item.id !== distractionId),
  };
}

export function focusActualMinutes(session: FocusSessionState, now = Date.now()) {
  const elapsed = Math.min(phaseDurationMs(session), timerElapsedMs(session, now));
  return Math.max(0, Math.round(elapsed / MINUTE));
}

export function beginBreak(
  session: FocusSessionState,
  actualMinutes: number,
  now = Date.now(),
): FocusSessionState {
  return {
    ...session,
    phase: 'break',
    phaseStartedAt: now,
    elapsedMs: 0,
    runningSince: now,
    focusLogged: true,
    focusActualMinutes: Math.max(0, Math.round(actualMinutes)),
  };
}

export function breakActualMinutes(session: FocusSessionState, now = Date.now()) {
  if (session.phase !== 'break') return 0;
  const elapsed = Math.min(phaseDurationMs(session), timerElapsedMs(session, now));
  return Math.max(0, Math.round(elapsed / MINUTE));
}

export function formatTimer(ms: number) {
  const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
}

type FocusContextValue = {

  saveError: string | null;

const FocusContext = createContext<FocusContextValue | null>(null);

function historyId(focusSessionId: string) {
  return `session_${focusSessionId}`;
}

export function FocusProvider({ children }: PropsWithChildren) {
  const { ownerKey, workspace, readWorkspace, updateWorkspace, waitForSaves } = useWorkspace();
  const celebrate = useAchievement();
  const [ready, setReady] = useState(false);
  const [session, setSession] = useState<FocusSessionState | null>(null);
  const [saveError, setSaveError] = useState<string | null>(null);
  const sessionRef = useRef<FocusSessionState | null>(null);
  const ownerRef = useRef<string | null>(null);
  const saveQueue = useRef(createFocusWriteQueue<FocusSessionState>(async (owner, next) => {
    if (next) await focusStorage.save(owner, next);
    else await focusStorage.clear(owner);
  }));

  const lifecycle = useRef({ ownerKey, ready, epoch: 0 });
  if (lifecycle.current.ownerKey !== ownerKey) lifecycle.current.epoch++;
  lifecycle.current.ownerKey = ownerKey;
  lifecycle.current.ready = ready;
  const ownerEpoch = lifecycle.current.epoch;

  const commitSession = useCallback(
    async (next: FocusSessionState | null) => {
      const owner = ownerRef.current;
      const previous = sessionRef.current;
      sessionRef.current = next;
      setSession(next);
      if (!owner) return;

      try {
        // Keep optimistic timer feedback while saving, but never treat a
        // rejected storage write as durable. Restore the last known state so
        // the user can retry the action after storage recovers.
        await saveQueue.current.write(owner, next);
        if (ownerRef.current === owner && sessionRef.current === next) {
          setSaveError(null);
        }
      } catch (error) {
        if (ownerRef.current === owner && sessionRef.current === next) {
          sessionRef.current = previous;
          setSession(previous);
          setSaveError('Focus changes could not be saved. The last change was reversed. Try again.');
        }
        throw error;
      }
    },
    [],
  );

  const workspaceReady = Boolean(workspace);

  useEffect(() => {
    let cancelled = false;
    ownerRef.current = ownerKey;
    sessionRef.current = null;
    setSession(null);
    setSaveError(null);

    if (!ownerKey || !workspace) {
      setReady(false);
      return () => {
        cancelled = true;
      };
    }

    setReady(false);

    focusStorage
      .load(ownerKey)
      .then(async (stored) => {
        if (cancelled || ownerRef.current !== ownerKey) return;

        if (stored) {
          const missingTopic = stored.topicId
            ? !findTopic(workspace, stored.topicId)
            : false;
          const missingTask = stored.taskId
            ? !taskById(workspace, stored.taskId)
            : false;
          if ((!stored.topicId && !stored.taskId) || missingTopic || missingTask) {
            await focusStorage.clear(ownerKey);
            stored = null;
          }
        }

        sessionRef.current = stored;
        setSession(stored);
        setReady(true);
      })
      .catch(() => {
        if (cancelled || ownerRef.current !== ownerKey) return;
        sessionRef.current = null;
        setSession(null);
        setReady(true);
      });

    return () => {
      cancelled = true;
    };
  }, [ownerKey, workspaceReady]);

  useEffect(() => {
    const current = sessionRef.current;
    if (!current || !workspace) return;

    const missingTopic = current.topicId
      ? !findTopic(workspace, current.topicId)
      : false;
    const missingTask = current.taskId
      ? !taskById(workspace, current.taskId)
      : false;
    if (!missingTopic && !missingTask && (current.topicId || current.taskId)) return;

    void commitSession(null);
  }, [
    commitSession,
    session?.topicId,
    session?.taskId,
    workspace?.topics.length,
    workspace?.tasks.length,
  ]);

  const start = useCallback(
    async (input: StartInput) => {
      if (!ownerKey || ownerRef.current !== ownerKey || lifecycle.current.epoch !== ownerEpoch || !lifecycle.current.ready || sessionRef.current) return null;
      const currentWorkspace = readWorkspace(ownerKey);
      if (!currentWorkspace) return null;

      const task = input.taskId ? taskById(currentWorkspace, input.taskId) : null;
      if (input.taskId && (!task || task.status !== 'open')) return null;

      const requestedTopic = input.topicId ? findTopic(currentWorkspace, input.topicId) : null;
      if (input.topicId && !requestedTopic) return null;
      if (task?.topicId && input.topicId && task.topicId !== input.topicId) return null;

      const topic =
        requestedTopic ??
        (task?.topicId ? findTopic(currentWorkspace, task.topicId) : null);
      if (!topic && !task) return null;

      const next = createFocusSession({
        ...(topic ? { topicId: topic.id } : {}),
        ...(task ? { taskId: task.id } : {}),
        topicName: task?.title || topic!.name,
        subject: task?.subject || topic?.subject || 'Study task',
        plannedMinutes: input.plannedMinutes,
        breakMinutes: input.breakMinutes,
        launchMode: input.launchMode,
        tinyStep: input.tinyStep,
      });

      const ownsStart = () => ownerRef.current === ownerKey && lifecycle.current.epoch === ownerEpoch && sessionRef.current?.id === next.id;
      await commitSession(next);
      // A saved start belongs to the original owner and this exact optimistic
      // session. A later owner/load/discard must not receive its outcome.
      if (!ownsStart() || !lifecycle.current.ready || !readWorkspace(ownerKey)) return null;
      const recorded = updateWorkspace(draft => recordFocusOutcome(draft, next, 'started'), {
        ownerKey, canApply: current => {
          const currentTask = next.taskId ? taskById(current, next.taskId) : null;
          return (!next.taskId || !!currentTask && currentTask.status === 'open' && currentTask.topicId === task?.topicId) && (!next.topicId || !!findTopic(current, next.topicId));
        },
      });
      if (!recorded) {
        if (ownsStart()) await commitSession(null);
        return null;
      }
      return next;
    },
    [commitSession, ownerKey, ownerEpoch, readWorkspace, updateWorkspace],
  );

  const transform = useCallback(
    async (change: (current: FocusSessionState) => FocusSessionState) => {
      const current = sessionRef.current;
      if (!current) return;
      try {
        await commitSession(change(current));
      } catch {
        // Focus controls are often fire-and-forget presses. commitSession
        // restores the last known session and surfaces its saveError.
      }
    },
    [commitSession],
  );

  const value = useMemo<FocusContextValue>(
    () => ({
      ready,
      session,
      saveError,

      start,

      pause: () => transform((current) => pauseFocusTimer(current)),
      resume: () => transform((current) => resumeFocusTimer(current)),
      extendLaunch: () => transform((current) => extendFiveMinuteLaunch(current)),
      park: async (text) => {
        const current = sessionRef.current;
        if (!current) return;
        // The distraction input is cleared only after this write succeeds.
        // If device storage fails, its text stays available for a retry.
        await commitSession(addParkedDistraction(current, text));
      },
      removeParked: (id) =>
        transform((current) => removeParkedDistraction(current, id)),

      async completeFocus(effort, now = Date.now()) {
        const current = sessionRef.current;
        if (!current || current.phase !== 'focus' || current.focusLogged) return;

        const actualMinutes = focusActualMinutes(current, now);

        let completionApplied = false;
        const updated = updateWorkspace((draft) => {
          recordFocusOutcome(draft, current, 'finished', actualMinutes, now);
          if (current.topicId) {
            const stableHistoryId = historyId(current.id);
            if (draft.history.some((entry) => entry.id === stableHistoryId)) {
              completionApplied = true;
              return;
            }

            const topic = findTopic(draft, current.topicId);
            if (!topic) return;

            if (topic.status === 'new') topic.status = 'learning';
            if (actualMinutes >= 5) topic.postponements = 0;

            draft.history.push({
              id: stableHistoryId,
              topicId: current.topicId,
              at: now,
              kind: 'study',
              minutes: actualMinutes,
              grade: 0,
              effort,
              prompt: '',
              response: '',
              confidence: 'unsure',
              plannedMinutes: current.plannedFocusMinutes,
              startedAt: current.startedAt,
              endedAt: now,
              launchMode: current.launchMode,
              tinyStep: current.tinyStep,
            });

            if (current.taskId) {
              const task = taskById(draft, current.taskId);
              if (task) {
                recordTaskMinutes(task, actualMinutes, now);
              }
            }

            completionApplied = true;
            return;
          }

          if (current.taskId) {
            const task = taskById(draft, current.taskId);
            if (!task) return;

            if (task.lastFocusSessionId === current.id) {
              completionApplied = true;
              return;
            }

            recordTaskMinutes(task, actualMinutes, now);

            const focusDate = new Date(now);
            const focusDayKey = [
              focusDate.getFullYear(),
              String(focusDate.getMonth() + 1).padStart(2, '0'),
              String(focusDate.getDate()).padStart(2, '0'),
            ].join('-');
            task.focusMinutesForDay =
              task.focusDayKey === focusDayKey
                ? Math.min(1440, (task.focusMinutesForDay ?? 0) + actualMinutes)
                : Math.min(1440, actualMinutes);
            task.focusDayKey = focusDayKey;
            task.lastFocusSessionId = current.id;
            task.updatedAt = now;
            completionApplied = true;
          }
        });

        if (!updated || !completionApplied) {
          await commitSession(null);
          return;
        }

        // The durable workspace write must win before the recoverable focus
        // session advances to break. Topic sessions write history; standalone
        // task sessions update task minutes with an idempotency marker.
        await waitForSaves();
        await commitSession(beginBreak(current, actualMinutes, now));
        celebrate({ title: 'Focus complete!', detail: `${actualMinutes} focused minutes · ${current.topicName}` });
      },

      async completeBreak(now = Date.now()) {
        const current = sessionRef.current;
        if (!current || current.phase !== 'break') return;

        const minutes = breakActualMinutes(current, now);
        if (minutes > 0) {
          updateWorkspace((draft) => {
            const stableBreakAt = current.phaseStartedAt;
            const existing = draft.breaks.find((item) => item.at === stableBreakAt);
            if (existing) {
              existing.minutes = Math.max(existing.minutes, minutes);
              return;
            }
            draft.breaks.push({ at: stableBreakAt, minutes });
          });
          await waitForSaves();
        }

        await commitSession(null);
      },

      async skipBreak(now = Date.now()) {
        const current = sessionRef.current;
        if (current?.phase === 'break') {
          const minutes = breakActualMinutes(current, now);
          if (minutes > 0) {
            updateWorkspace((draft) => {
              const stableBreakAt = current.phaseStartedAt;
              const existing = draft.breaks.find((item) => item.at === stableBreakAt);
              if (existing) {
                existing.minutes = Math.max(existing.minutes, minutes);
                return;
              }
              draft.breaks.push({ at: stableBreakAt, minutes });
            });
            await waitForSaves();
          }
        }
        await commitSession(null);
      },

      async discard() {
        const current = sessionRef.current;
        if (current?.phase === 'focus') updateWorkspace(draft => recordFocusOutcome(draft, current, 'abandoned'));
        await commitSession(null);
      },
    }),
    [celebrate, commitSession, ready, saveError, session, start, transform, updateWorkspace, waitForSaves],
  );

  return <FocusContext.Provider value={value}>{children}</FocusContext.Provider>;
}

export function useFocus() {
  const value = useContext(FocusContext);
  if (!value) throw new Error('useFocus must be used inside FocusProvider.');
  return value;
}

// Pure dial sizing for the Focus and Break screens.
// 24dp extra width reserves space for the 10dp outer ring on each side.
export const FOCUS_DIAL_MIN_SIZE = 128;
export const FOCUS_DIAL_MAX_SIZE = 228;
const DIAL_EDGE_RESERVE = 24;
const PROGRESS_MARKS = 40;

export function focusDialSize(windowWidth: number, horizontalPadding: number): number {
  if (!Number.isFinite(windowWidth) || !Number.isFinite(horizontalPadding)) {
    return FOCUS_DIAL_MIN_SIZE;
  }
  const available = Math.floor(windowWidth - 2 * horizontalPadding - DIAL_EDGE_RESERVE);
  return Math.max(FOCUS_DIAL_MIN_SIZE, Math.min(FOCUS_DIAL_MAX_SIZE, available));
}

export function focusDialMarkPosition(size: number, index: number) {
  const angle = index * Math.PI * 2 / PROGRESS_MARKS;
  const radius = size / 2 - 11;
  return {
    left: size / 2 - 1.5 + Math.sin(angle) * radius,
    top: size / 2 - 4.5 - Math.cos(angle) * radius,
  };
}

import { WorkspaceState } from '@/src/data/schema';
import { buildReviewQueue } from '@/src/features/review/review-engine';
import { findTopic } from '@/src/features/topics/topic-operations';

export const POST_FOCUS_RETRIEVAL_LIMIT = 3 as const;
export const MIN_POST_FOCUS_RETRIEVAL_MINUTES = 5 as const;

export type PostFocusRetrievalSuggestion = {
  topicId: string;
  topicName: string;
  subject: string;
  cardIds: string[];
  count: number;
};

export function buildPostFocusRetrievalSuggestion(
  workspace: WorkspaceState,
  topicId: string | undefined,
  focusMinutes: number,
  now = Date.now(),
): PostFocusRetrievalSuggestion | null {
  if (!topicId || focusMinutes < MIN_POST_FOCUS_RETRIEVAL_MINUTES) return null;

  const topic = findTopic(workspace, topicId);
  if (!topic || topic.cards.length === 0) return null;

  const queue = buildReviewQueue(workspace, {
    now,
    topicId,
    limit: POST_FOCUS_RETRIEVAL_LIMIT,
  });

  if (!queue.length) return null;

  return {
    topicId: topic.id,
    topicName: topic.name,
    subject: topic.subject,
    cardIds: queue.map((item) => item.cardId),
    count: queue.length,
  };
}
