// Execute the actual screen handler with delayed/rejected storage, rather
// than treating a legacy handler name as proof of the save-before-route flow.
async function verifyBreakTransitions() {
  const screen = fs.readFileSync('app/(app)/focus/index.tsx', 'utf8');
  const source = ts.createSourceFile('focus.tsx', screen, ts.ScriptTarget.Latest, true, ts.ScriptKind.TSX);
  let handler: ts.FunctionDeclaration | undefined;
  function visit(node: ts.Node) {
    if (ts.isFunctionDeclaration(node) && node.name?.text === 'runBreakTransition') handler = node;
    ts.forEachChild(node, visit);
  }
  visit(source);
  assert.ok(handler, 'The Focus screen must have a shared break exit handler.');
  const executable = ts.transpileModule(handler.getText(source), {
    compilerOptions: { target: ts.ScriptTarget.ES2022, module: ts.ModuleKind.None },
  }).outputText;

  function simulate(phase = 'break') {
    const routes: unknown[] = [];
    const saves: string[] = [];
    const lock = { current: false };
    let busy: string | null = null;
    let error = '';
    let persist: () => Promise<void> = async () => {};
    const bindings = {
      session: { phase },
      breakTransitionLock: lock,
      retrievalSuggestion: { topicId: 'topic_light', count: 3 } as { topicId: string; count: number } | null,
      setBreakTransition: (value: string | null) => { busy = value; },
      setFocusError: (value: string) => { error = value; },
      completeBreak: async () => { saves.push('complete'); await persist(); },
      skipBreak: async () => { saves.push('skip'); await persist(); },
      router: { replace: (route: unknown) => { routes.push(JSON.parse(JSON.stringify(route))); } },
      Date: { now: () => now },
    };
    const run = vm.runInNewContext(executable + '\nrunBreakTransition;', bindings, { timeout: 1000 }) as
      (action: 'finish' | 'retrieve' | 'skip') => Promise<void>;
    return { bindings, routes, saves, lock, run,
      setPersistence: (next: () => Promise<void>) => { persist = next; },
      busy: () => busy, error: () => error };
  }

  const rapid = simulate();
  let release!: () => void;
  rapid.setPersistence(() => new Promise<void>(resolve => { release = resolve; }));
  const first = rapid.run('retrieve');
  assert.equal(rapid.lock.current, true);
  assert.equal(rapid.busy(), 'retrieve');
  assert.deepEqual(rapid.routes, [], 'Do not navigate before break storage succeeds.');
  await rapid.run('skip');
  assert.deepEqual(rapid.saves, ['complete'], 'A second rapid exit must not start another save.');
  rapid.bindings.retrievalSuggestion = { topicId: 'different_topic', count: 1 };
  release();
  await first;
  assert.deepEqual(rapid.routes, [{ pathname: '/(app)/review', params: {
    topicId: 'topic_light', autoStart: '1', limit: '3', source: 'focus',
  } }], 'Retrieve the bounded topic captured at the tap, even if the live suggestion changes.');
  await rapid.run('finish');
  assert.equal(rapid.routes.length, 1, 'Keep duplicate exits blocked through successful navigation.');

  const retry = simulate();
  retry.setPersistence(async () => { throw new Error('device storage unavailable'); });
  await retry.run('finish');
  assert.deepEqual(retry.routes, [], 'A rejected save must keep the user on Break.');
  assert.equal(retry.bindings.session.phase, 'break');
  assert.equal(retry.lock.current, false);
  assert.equal(retry.busy(), null);
  assert.match(retry.error(), /could not be saved.*retry/i);
  retry.setPersistence(async () => {});
  await retry.run('finish');
  assert.deepEqual(retry.saves, ['complete', 'complete']);
  assert.deepEqual(retry.routes, ['/(app)/today']);
  assert.equal(retry.error(), '', 'A successful retry clears the previous failure.');

  const noRecall = simulate();
  noRecall.bindings.retrievalSuggestion = null;
  await noRecall.run('retrieve');
  assert.deepEqual(noRecall.saves, ['complete']);
  assert.deepEqual(noRecall.routes, ['/(app)/today']);

  const skip = simulate();
  skip.setPersistence(() => new Promise<void>(resolve => { release = resolve; }));
  const skipping = skip.run('skip');
  assert.deepEqual(skip.saves, ['skip']);
  assert.deepEqual(skip.routes, []);
  release();
  await skipping;
  assert.deepEqual(skip.routes, ['/(app)/today']);

  const wrongPhase = simulate('focus');
  await wrongPhase.run('retrieve');
  assert.deepEqual(wrongPhase.saves, []);
  assert.deepEqual(wrongPhase.routes, []);
  console.log('✓ Actual break handler blocks rapid exits, waits for saves, preserves recall and recovers from failure');
}

import { RefObject, useCallback, useEffect, useRef, useState } from 'react';
import { Keyboard, ScrollView, View } from 'react-native';
import { useNavigation, useFocusEffect, useLocalSearchParams } from 'expo-router';
import type { NavigationProp } from 'expo-router/react-navigation';
import { GuideStep, TutorialAnchor, TutorialId } from './guide-registry';
import { measureGuideView } from './measure-guide-view';

/** A virtualized list can provide the same measured viewport/offset behavior. */
export type GuideScrollAdapter = {
  measureNode: () => Pick<View, 'measureInWindow'> | null;
  scrollTo: (y: number) => void;
};

/** Shared presentation state only: never starts, pauses or completes study sessions. */
export function useScreenGuide<A extends TutorialAnchor>(ready: boolean, ownerKey: string | null,
  tutorialId: TutorialId, anchors: Record<A, RefObject<View | null>>, steps: readonly GuideStep<A>[],
  scrollAdapter?: GuideScrollAdapter) {
  const navigation = useNavigation<NavigationProp<Record<string, { guide?: string | string[] }>>>();
  const { guide: guideParam } = useLocalSearchParams<{ guide?: string | string[] }>();
  const guide = Array.isArray(guideParam) ? guideParam[0] : guideParam;
  const [open, setOpen] = useState(false);
  const openRef = useRef(false);
  const scroll = useRef<ScrollView>(null);
  const offset = useRef(0);
  const originalOffset = useRef(0);
  const start = useCallback(() => {
    if (openRef.current || !ready) return;
    Keyboard.dismiss();
    originalOffset.current = offset.current;
    openRef.current = true;
    setOpen(true);
  }, [ready]);
  const close = useCallback(() => {
    if (!openRef.current) return;
    openRef.current = false;
    setOpen(false);
    if (scrollAdapter) scrollAdapter.scrollTo(originalOffset.current);
    else scroll.current?.scrollTo({ y: originalOffset.current, animated: false });
  }, [scrollAdapter]);
  useFocusEffect(useCallback(() => () => close(), [close]));
  useEffect(() => { close(); }, [ownerKey, close]);
  useEffect(() => {
    if (!ready) { close(); return; }
    if (guide === tutorialId) {
      // Consume on this screen, including inside a nested Library stack.
      navigation.setParams({ guide: undefined });
      start();
    }
  }, [guide, ready, close, start, tutorialId, navigation]);

  const revealAnchor = useCallback(async (anchor: A, shouldContinue: () => boolean = () => true) => {
    const [target, viewport] = await Promise.all([
      measureGuideView(anchors[anchor].current), measureGuideView(scrollAdapter ? scrollAdapter.measureNode() : scroll.current?.getNativeScrollRef() ?? null),
    ]);
    if (!openRef.current || !shouldContinue() || !target || !viewport) return;
    // Scrolling is immediate even when motion is enabled: the spotlight never chases a moving target.
    const y = Math.max(0, offset.current + target.y - viewport.y - 16);
    if (scrollAdapter) scrollAdapter.scrollTo(y);
    else scroll.current?.scrollTo({ y, animated: false });
    await new Promise<void>(resolve => setTimeout(resolve, 100));
  }, [anchors, scrollAdapter]);

  return { open, start, close, anchors, steps, scroll, offset, revealAnchor };
}

import { RefObject, useEffect, useRef, useState } from 'react';
import { AccessibilityInfo, Modal, Pressable, ScrollView, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { useReducedMotion } from '@/src/features/accessibility/useReducedMotion';
import { Reveal } from '@/src/components/motion/Reveal';
import { colors, radius } from '@/src/theme/tokens';
import { GuideStep, TutorialAnchor } from './guide-registry';
import { GuideRect, guideCardMaxHeight, guideCardTop, spotlightRect } from './guide-geometry';
import { measureGuideView } from './measure-guide-view';

type Props<A extends TutorialAnchor> = {
  title?: string;
  steps: readonly GuideStep<A>[];
  anchors: Record<A, RefObject<View | null>>;
  revealAnchor: (anchor: A, shouldContinue?: () => boolean) => Promise<void>;
  onClose: () => void;
};

export function ScreenGuide<A extends TutorialAnchor>({ title = 'Today', steps, anchors, revealAnchor, onClose }: Props<A>) {
  const [index, setIndex] = useState(0);
  const [shown, setShown] = useState(false);
  const [target, setTarget] = useState<GuideRect | null>(null);
  const [measuring, setMeasuring] = useState(true);
  const [retry, setRetry] = useState(0);
  const [cardHeight, setCardHeight] = useState(240);
  const [hostSize, setHostSize] = useState({ width: 0, height: 0 });
  const hostRef = useRef<View>(null);
  const reduced = useReducedMotion();
  const dimensions = useWindowDimensions();
  const insets = useSafeAreaInsets();
  const step = steps[Math.min(index, steps.length - 1)];
  const width = hostSize.width || dimensions.width;
  const height = hostSize.height || dimensions.height;

  useEffect(() => {
    // Wait for the modal layout before scrolling; a superseded measurement must not move the screen.
    if (!shown || !step || hostSize.width <= 0 || hostSize.height <= 0) return;
    let cancelled = false;
    setMeasuring(true); setTarget(null);
    void (async () => {
      try {
        await revealAnchor(step.anchor, () => !cancelled);
        if (cancelled) return;
        const [anchor, host] = await Promise.all([measureGuideView(anchors[step.anchor].current), measureGuideView(hostRef.current)]);
        if (cancelled) return;
        setTarget(anchor && host ? spotlightRect(anchor, host) : null);
        AccessibilityInfo.announceForAccessibility(`Step ${index + 1} of ${steps.length}. ${step.title}. ${step.explanation}`);
      } catch {
        if (!cancelled) setTarget(null);
      } finally { if (!cancelled) setMeasuring(false); }
    })();
    return () => { cancelled = true; };
  }, [shown, step, index, steps.length, anchors, revealAnchor, dimensions.width, dimensions.height, dimensions.fontScale, retry, hostSize.width, hostSize.height]);

  if (!step) return null;
  const maxHeight = guideCardMaxHeight(target, height, insets.top, insets.bottom);
  const top = guideCardTop(target, height, Math.min(cardHeight, maxHeight), insets.top, insets.bottom);
  return <Modal transparent visible animationType={reduced ? 'none' : 'fade'} statusBarTranslucent onShow={() => setShown(true)} onRequestClose={onClose}>
    <View ref={hostRef} collapsable={false} style={styles.overlay} onLayout={event => setHostSize(event.nativeEvent.layout)}>
      <View pointerEvents="none" style={StyleSheet.absoluteFill} accessible={false} importantForAccessibility="no-hide-descendants">
        {target ? <>
          <View style={[styles.shade, { top: 0, left: 0, right: 0, height: target.y }]} />
          <View style={[styles.shade, { top: target.y, left: 0, width: target.x, height: target.height }]} />
          <View style={[styles.shade, { top: target.y, left: target.x + target.width, right: 0, height: target.height }]} />
          <View style={[styles.shade, { top: target.y + target.height, left: 0, right: 0, bottom: 0 }]} />
          <View testID="guide-spotlight" style={[styles.ring, { left: target.x, top: target.y, width: target.width, height: target.height }]} />
        </> : <View style={[styles.shade, StyleSheet.absoluteFill]} />}
      </View>
      <View testID="guide-card" accessibilityViewIsModal style={[styles.card, { top, left: Math.max(12, (width - 420) / 2), width: Math.min(width - 24, 420), maxHeight }]} onLayout={event => setCardHeight(event.nativeEvent.layout.height)}>
        <View style={styles.heading}>
          <Text style={styles.step}>{title.toUpperCase()} · {index + 1}/{steps.length}</Text>
          <Pressable accessibilityRole="button" accessibilityLabel="Skip tutorial" onPress={onClose} style={styles.skip}><Text style={styles.secondaryText}>Skip</Text></Pressable>
        </View>
        <ScrollView style={styles.copyScroll} contentContainerStyle={styles.copy}>
          <Reveal key={step.anchor} distance={4} duration={160}>
            <Text accessibilityRole="header" style={styles.title}>{step.title}</Text>
            <Text style={styles.explanation}>{step.explanation}</Text>
            {!measuring && !target ? <Text accessibilityLiveRegion="polite" style={styles.fallback}>This control is not visible right now. Retry its highlight, or continue the explanation.</Text> : null}
          </Reveal>
        </ScrollView>
        {!measuring && !target ? <Pressable accessibilityRole="button" onPress={() => setRetry(v => v + 1)} style={styles.retry}><Text style={styles.secondaryText}>Retry highlight</Text></Pressable> : null}
        <View style={styles.actions}>
          <Pressable accessibilityRole="button" accessibilityLabel="Previous tutorial step" disabled={index === 0} onPress={() => setIndex(i => Math.max(0, i - 1))} style={[styles.back, index === 0 && styles.disabled]}><Text style={styles.secondaryText}>Back</Text></Pressable>
          <Pressable accessibilityRole="button" accessibilityLabel={index === steps.length - 1 ? 'Finish tutorial' : 'Next tutorial step'} onPress={() => index === steps.length - 1 ? onClose() : setIndex(i => i + 1)} style={styles.next}><Text style={styles.nextText}>{index === steps.length - 1 ? 'Done' : 'Next'}</Text></Pressable>
        </View>
      </View>
    </View>
  </Modal>;
}

import { useCallback, useEffect, useMemo, useState } from 'react';
import { createGuideOfferController, emptyGuideOfferState, GuideOfferChoice } from './guide-offer-state';
import { guideOfferStorage } from './guide-offer-storage';
import type { TutorialId } from './guide-registry';

export function useGuideOffer(tutorialId: TutorialId, ownerKey: string | null, eligible: boolean, tutorialOpen: boolean) {
  const [state, setState] = useState(() => ({ ...emptyGuideOfferState(), tutorialId }));
  const controller = useMemo(() => createGuideOfferController(guideOfferStorage, tutorialId, next => setState({ ...next, tutorialId })), [tutorialId]);
  const currentOwner = !!ownerKey && state.ownerKey === ownerKey && state.tutorialId === tutorialId;

  useEffect(() => {
    void controller.activate(ownerKey);
    return () => controller.deactivate();
  }, [controller, ownerKey]);
  useEffect(() => {
    // Explicit header/Help replay also counts as seeing this screen's guide.
    if (currentOwner && tutorialOpen && !state.choice) void controller.choose(ownerKey, 'shown');
  }, [controller, currentOwner, ownerKey, state.choice, tutorialOpen]);

  const choose = useCallback((choice: GuideOfferChoice) => {
    void controller.choose(ownerKey, choice);
  }, [controller, ownerKey]);
  const retry = useCallback(() => {
    if (state.choice) void controller.choose(ownerKey, state.choice);
  }, [controller, ownerKey, state.choice]);

  return {
    visible: currentOwner && state.ready && !state.choice && eligible && !tutorialOpen,
    saveFailed: currentOwner && state.saveFailed,
    choose,
    retry,
  };
}

  </View>;
}
